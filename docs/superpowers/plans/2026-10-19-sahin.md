```python
#!/usr/bin/env python3
"""Fails when a tree holds what Sahin must never ship: Windows executables we didn't build,
game data, publisher media or Microsoft redistributables.

  tools/audit.py ROOT [ROOT...]

ROOT/.audit-allow lists globs (relative to ROOT) of files built from source that may be PE
files (Wine's DLLs, DXVK). Exit 0 clean, 1 with one line per finding.
"""
import fnmatch, pathlib, sys

BANNED_NAMES = ["blur.exe", "speed.exe", "binkw32.dll", "d3dx9_*.dll", "d3dcompiler_*.dll",
                "msvcr*.dll", "msvcp*.dll", "xinput1_*.dll", "physx*.dll", "*.securom"]
BANNED_EXT = {".bik", ".bk2", ".at9", ".pak", ".sig", ".level", ".levelphysics",
              ".levelrender", ".levelstream", ".asi", ".cab"}
SKIP_DIRS = {".git", "prospero-win", "PS5_Mesa", "PS5_Vulkan", "PS5_PayloadSDK",
             "ps5-native-app-boilerplate", ".deps", "build", "node_modules"}


def is_pe(path):
    """Any MZ file counts: DOS stubs and truncated PEs are executables too."""
    try:
        with open(path, "rb") as f:
            return f.read(2) == b"MZ"
    except OSError:
        return False


def audit(root):
    root = pathlib.Path(root)
    allow_file = root / ".audit-allow"
    allow = allow_file.read_text().split() if allow_file.exists() else []
    findings = []
    for path in sorted(root.rglob("*")):
        rel = path.relative_to(root)
        if any(part in SKIP_DIRS for part in rel.parts) or not path.is_file():
            continue
        name = path.name.lower()
        if any(fnmatch.fnmatch(str(rel), glob) for glob in allow):
            continue
        if any(fnmatch.fnmatch(name, glob) for glob in BANNED_NAMES):
            findings.append(f"{rel}: banned file name")
        elif path.suffix.lower() in BANNED_EXT:
            findings.append(f"{rel}: game data or media type")
        elif is_pe(path):
            findings.append(f"{rel}: Windows executable not in .audit-allow")
    return findings


def main():
    findings = [f for root in sys.argv[1:] or ["."] for f in audit(root)]
    for line in findings:
        print(line)
    sys.exit(1 if findings else 0)


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Records a Blur copy's data files (path, size, CRC-32, SHA-256) as profile/blur/base.json.

  profile/blur/scan-copy.py GAME_FOLDER OUT.json

Only numbers are written. Executables, mods and the installer's leftovers are skipped: the
plan accepts any exe, and mods in a copy are left out of the install.
"""
import hashlib, json, pathlib, sys, zlib

SKIP_TOP = {"blur.exe", "dsound.dll", "dinput8.dll", "d3d9.dll", "lua5.1.dll",
            "discord-rpc.dll", "blur_rpc.log", "scripts", "amax", "uninstall", "redist",
            "docs", "5.ico", "firewallinstallhelper.dll"}


def digest(path):
    sha, crc = hashlib.sha256(), 0
    with open(path, "rb") as f:
        while chunk := f.read(1 << 20):
            sha.update(chunk)
            crc = zlib.crc32(chunk, crc)
    return sha.hexdigest(), f"{crc & 0xffffffff:08x}"


def main(game, out):
    game = pathlib.Path(game)
    files = []
    for path in sorted(game.rglob("*")):
        rel = path.relative_to(game)
        if not path.is_file() or rel.parts[0].lower() in SKIP_TOP:
            continue
        sha, crc = digest(path)
        files.append({"path": rel.as_posix(), "size": path.stat().st_size, "crc32": crc, "sha256": sha})
    pathlib.Path(out).write_text(json.dumps({"files": files}, indent=1) + "\n")


if __name__ == "__main__":
    main(*sys.argv[1:3])
```

```python
#!/usr/bin/env python3
"""Draws Sahin's Home-screen art (our own, nothing from the game): icon, background, pic0.

  art/make-art.py [OUT_DIR]   (default: art/)
"""
import pathlib, random, sys
from PIL import Image, ImageDraw, ImageFilter

COLOURS = [(45, 226, 230), (255, 60, 172), (255, 176, 0)]


def gradient(w, h):
    img = Image.new("RGB", (w, h))
    px = img.load()
    for y in range(h):
        t = y / (h - 1)
        c = (int(11 * (1 - t)), int(15 * (1 - t)), int(42 * (1 - t) + 6 * t))
        for x in range(w):
            px[x, y] = c
    return img


def streaks(img, rng, count):
    w, h = img.size
    glow = Image.new("RGBA", img.size, (0, 0, 0, 0))
    sharp = Image.new("RGBA", img.size, (0, 0, 0, 0))
    g, s = ImageDraw.Draw(glow), ImageDraw.Draw(sharp)
    for _ in range(count):
        y = rng.randrange(h)
        x = rng.randrange(-200, w)
        length = rng.randrange(w // 6, w)
        colour = rng.choice(COLOURS) + (rng.randrange(40, 200),)
        width = rng.randrange(4, 14)
        g.line([(x, y), (x + length, y)], fill=colour, width=width * 2)
        s.line([(x, y), (x + length, y)], fill=colour[:3] + (255,), width=2)
    out = img.convert("RGBA")
    out.alpha_composite(glow.filter(ImageFilter.GaussianBlur(6)))
    out.alpha_composite(sharp)
    return out


def falcon(img):
    w, h = img.size
    d = ImageDraw.Draw(img)
    cx, cy, s = w / 2, h / 2, min(w, h) / 3
    left = [(cx, cy + s * 0.35), (cx - s, cy - s * 0.45), (cx - s * 0.35, cy - s * 0.2)]
    right = [(cx, cy + s * 0.35), (cx + s, cy - s * 0.45), (cx + s * 0.35, cy - s * 0.2)]
    glow = Image.new("RGBA", img.size, (0, 0, 0, 0))
    gd = ImageDraw.Draw(glow)
    for poly in (left, right):
        gd.polygon(poly, fill=(45, 226, 230, 200))
    img.alpha_composite(glow.filter(ImageFilter.GaussianBlur(12)))
    for poly in (left, right):
        d.polygon(poly, fill=(255, 255, 255, 255))
    return img


def main(out):
    out = pathlib.Path(out)
    out.mkdir(parents=True, exist_ok=True)
    rng = random.Random(1923)
    icon = falcon(streaks(gradient(512, 512), rng, 18))
    icon.convert("RGB").save(out / "sahin.png", optimize=False)
    streaks(gradient(1920, 1080), rng, 40).convert("RGB").save(out / "sahin-bg.png", optimize=False)
    falcon(streaks(gradient(1920, 1080), rng, 40)).convert("RGB").save(out / "sahin-pic0.png", optimize=False)


if __name__ == "__main__":
    main(sys.argv[1] if len(sys.argv) > 1 else pathlib.Path(__file__).parent)
```
